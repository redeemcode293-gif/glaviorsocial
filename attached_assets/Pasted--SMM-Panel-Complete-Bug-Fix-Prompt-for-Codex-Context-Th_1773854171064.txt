## CRITICAL BUG #1 — INR Price Corruption (₹350 showing as $35,00,000)

### Root Cause

Provider APIs return prices **in Indian Rupees (₹) per 1000 units**. The current code removes commas (correct) but **never divides by 92** (the INR→USD rate). So ₹350 gets stored as $350 instead of $3.80, and then the regional multiplier makes it show as $35,00,000 on the panel.

### Fix required in: `src/components/admin/BulkServiceImport.tsx`

Replace the existing price parsing functions with these exact implementations:

```typescript
/** 1 INR = 1/92 USD */
const INR_TO_USD = 1 / 92;

/** Any panel price above this in USD is a sure sign of un-converted INR */
const MAX_SANE_USD = 500;

/**
 * parseRawPrice — handles INR lakh format (1,00,000.50), INR thousand format
 * (15,650.19), plain USD (0.50), strips all currency symbols.
 * KEY FIX: removes ALL commas before parseFloat so "15,650.19" → 15650.19
 * (NOT 15.19 as some naive implementations do).
 */
function parseRawPrice(raw: string | number): number {
  if (typeof raw === "number") return isNaN(raw) ? 0 : raw;
  if (!raw) return 0;
  let cleaned = String(raw).replace(/[₹Rs$€£\s]/gi, "").trim();
  const dotParts = cleaned.split(".");
  if (dotParts.length > 2) {
    const intPart = dotParts.slice(0, -1).join("").replace(/,/g, "");
    cleaned = intPart + "." + dotParts[dotParts.length - 1];
  } else {
    cleaned = cleaned.replace(/,/g, ""); // remove ALL commas
  }
  const parsed = parseFloat(cleaned);
  return isNaN(parsed) ? 0 : parsed;
}

/**
 * toUSD — converts provider price to USD.
 * - If currency is INR → divide by 92
 * - If value > 100 with any currency → heuristic: treat as INR, divide by 92
 * - Otherwise → treat as USD
 */
function toUSD(raw: string | number, currency: string): number {
  const value = parseRawPrice(raw);
  if (value <= 0 || isNaN(value)) return 0;
  const cur = (currency || "USD").toUpperCase().trim();
  if (cur === "INR" || cur === "₹" || cur === "RS") {
    return value * INR_TO_USD;
  }
  if (value > 100) {
    console.warn(`[BulkImport] price ${value} > $100 for currency "${cur}" — treating as INR`);
    return value * INR_TO_USD;
  }
  return value;
}
```

**Also add** a `providerCurrency` state (default `"INR"`) and a currency selector UI in the connection form:

**Implement `fixCorruptedPrices()`:**
```typescript
const fixCorruptedPrices = async () => {
  setIsFixingPrices(true);
  try {
    // Fetch all corrupted services in pages
    const corrupted: Service[] = [];
    for (let offset = 0; ; offset += 1000) {
      const { data } = await supabase
        .from("services")
        .select("*")
        .gt("base_price", CORRUPTED_PRICE_THRESHOLD)
        .order("id")
        .range(offset, offset + 999);
      if (!data || !data.length) break;
      corrupted.push(...data);
      if (data.length < 1000) break;
    }
    setPriceFixProgress({ done: 0, total: corrupted.length });

    for (let i = 0; i < corrupted.length; i += 50) {
      const batch = corrupted.slice(i, i + 50);
      for (const svc of batch) {
        const fixedBase = Number(svc.base_price) * INR_TO_USD;
        const fixedProvider = svc.provider_price !== null ? Number(svc.provider_price) * INR_TO_USD : null;
        await supabase.from("services").update({ base_price: fixedBase, provider_price: fixedProvider }).eq("id", svc.id);
        await supabase.from("panel_services").update({ price: fixedBase }).eq("service_id", svc.service_id);
      }
      setPriceFixProgress({ done: Math.min(i + 50, corrupted.length), total: corrupted.length });
      await new Promise(r => setTimeout(r, 80));
    }

    toast({ title: `✓ Fixed ${corrupted.length} services` });
    await fetchData();
  } catch (err) {
    toast({ title: "Fix failed", description: String(err), variant: "destructive" });
  }
  setIsFixingPrices(false);
};
```

## BUG #2 — Missing `api_keys` RLS Policies

### Fix required in: New migration file

The `api_keys` table is missing INSERT and DELETE RLS policies, which causes "Failed to regenerate API key" errors.

Create file: `supabase/migrations/20260318200000_fix_all_prices_and_policies.sql`

```sql
-- Step 1: Add currency column to api_providers
ALTER TABLE public.api_providers
  ADD COLUMN IF NOT EXISTS currency TEXT DEFAULT 'INR';

-- Step 2: Fix corrupted INR prices in services (base_price > $50 means raw INR)
UPDATE public.services
SET
  base_price = base_price / 92,
  provider_price = CASE WHEN provider_price IS NOT NULL THEN provider_price / 92 ELSE NULL END
WHERE base_price > 50;

-- Step 3: Fix panel_services prices too
UPDATE public.panel_services
SET price = price / 92
WHERE price > 50;

-- Step 4: Missing api_keys INSERT policy
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_policies WHERE tablename = 'api_keys'
      AND policyname = 'Users can insert their own api keys'
  ) THEN
    EXECUTE '
      CREATE POLICY "Users can insert their own api keys" ON public.api_keys
        FOR INSERT WITH CHECK (auth.uid() = user_id)
    ';
  END IF;
END $$;

-- Step 5: Missing api_keys DELETE policy
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_policies WHERE tablename = 'api_keys'
      AND policyname = 'Users can delete their own api keys'
  ) THEN
    EXECUTE '
      CREATE POLICY "Users can delete their own api keys" ON public.api_keys
        FOR DELETE USING (auth.uid() = user_id)
    ';
  END IF;
END $$;

-- Step 6: Fix wallets INSERT policy (broken for new user signups)
DROP POLICY IF EXISTS "Users can insert own wallet" ON public.wallets;
DROP POLICY IF EXISTS "System can insert wallets" ON public.wallets;
CREATE POLICY "Users can insert own wallet" ON public.wallets
  FOR INSERT WITH CHECK (auth.uid() = user_id);

-- Step 7: Fix orders status constraint — add 'failed' and 'partial'
ALTER TABLE public.orders DROP CONSTRAINT IF EXISTS orders_status_check;
ALTER TABLE public.orders ADD CONSTRAINT orders_status_check
  CHECK (status IN (
    'pending','processing','in_progress','completed',
    'partial','cancelled','refunded','failed'
  ));

-- Step 8: Performance indexes
CREATE INDEX IF NOT EXISTS idx_orders_status         ON public.orders(status);
CREATE INDEX IF NOT EXISTS idx_orders_user_id        ON public.orders(user_id);
CREATE INDEX IF NOT EXISTS idx_orders_provider_order ON public.orders(provider_order_id);
CREATE INDEX IF NOT EXISTS idx_services_base_price   ON public.services(base_price);
CREATE INDEX IF NOT EXISTS idx_panel_services_price  ON public.panel_services(price);
```

## Verification

After all fixes, prices should satisfy:
- `services.base_price` range: `$0.0001 – $10.00` for typical services
- `panel_services.price` range: same
- No service should show price above `$50` unless it's genuinely a premium/expensive service
- Provider balance stored in USD (e.g., ₹5000 balance → stored as ~$54)

The price flow is:
```
Provider raw price (INR, e.g. ₹350/1K)
  ÷ 92 → USD provider price ($3.80/1K)
  × (1 + margin) → USD base price ($4.94/1K at 30% margin)
  × regional_multiplier → displayed price (e.g. 1.0× → $4.94)
```

NOT:
```
₹350 stored as $350 → × 1.3 margin → $455 → × 1000 regional multiplier → $455,000  ← BUG
```
